cmake_minimum_required(VERSION 3.16)
project(go2_navfn_validation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
if(NOT EXISTS "${GO2_NAVFN_SOURCE}/src/navfn.cpp")
  message(FATAL_ERROR "Set GO2_NAVFN_SOURCE to an unmodified Humble nav2_navfn_planner source directory")
endif()
add_executable(grid_plan grid_plan.cpp "${GO2_NAVFN_SOURCE}/src/navfn.cpp")
if(NOT EXISTS "${GO2_NAVFN_SOURCE}/../nav2_costmap_2d/include/nav2_costmap_2d/denoise/image_processing.hpp")
  message(FATAL_ERROR "The Humble Nav2 checkout must include the upstream DenoiseLayer algorithm")
endif()
target_include_directories(grid_plan PRIVATE "${GO2_NAVFN_SOURCE}/include"
  "${GO2_NAVFN_SOURCE}/../nav2_costmap_2d/include")
ament_target_dependencies(grid_plan rclcpp)
